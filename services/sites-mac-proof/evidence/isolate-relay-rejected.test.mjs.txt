import { test } from 'node:test';
import assert from 'node:assert/strict';
import { readFile } from 'node:fs/promises';
import { Miniflare } from 'miniflare';
const A='aaaaaaaa-aaaa-4aaa-8aaa-aaaaaaaaaaaa', B='bbbbbbbb-bbbb-4bbb-8bbb-bbbbbbbbbbbb';
const bindings={PROOF_CONTROL_TOKEN:'control-test-token-32-bytes-minimum',PROOF_DEVICE_A_TOKEN:'device-a-test-token-32-bytes-minimum',PROOF_DEVICE_B_TOKEN:'device-b-test-token-32-bytes-minimum'};
async function fixture(t){const mf=new Miniflare({modules:true,script:await readFile(new URL('../worker/index.js',import.meta.url),'utf8'),compatibilityDate:'2026-07-31',bindings});t.after(()=>mf.dispose());return mf;}
function open(mf,device,token){return mf.dispatchFetch('http://proof/ws?device='+device,{headers:{Upgrade:'websocket','x-localoud-proof-token':token}});}
function invoke(mf,device,nonce='1234567890abcdef1234567890abcdef'){return mf.dispatchFetch('http://proof/api/invoke',{method:'POST',headers:{'content-type':'application/json','x-localoud-proof-token':bindings.PROOF_CONTROL_TOKEN},body:JSON.stringify({device_id:device,request_id:crypto.randomUUID(),nonce})});}
function responder(ws,label){ws.accept();ws.addEventListener('message',event=>{const m=JSON.parse(event.data);if(m.type==='probe')ws.send(JSON.stringify({type:'result',request_id:m.request_id,nonce:m.nonce,label}));});}
test('upgrade requires the device capability and rejects a different device token',async t=>{const mf=await fixture(t);assert.equal((await open(mf,A,'wrong')).status,403);assert.equal((await open(mf,B,bindings.PROOF_DEVICE_A_TOKEN)).status,403);assert.equal((await mf.dispatchFetch('http://proof/ws?device='+A)).status,426);});
test('HTTP probe reaches only its selected socket and returns matching nonce',async t=>{const mf=await fixture(t);const a=await open(mf,A,bindings.PROOF_DEVICE_A_TOKEN),b=await open(mf,B,bindings.PROOF_DEVICE_B_TOKEN);assert.equal(a.status,101);assert.equal(b.status,101);responder(a.webSocket,'A');responder(b.webSocket,'B');let observedA=0,observedB=0;a.webSocket.addEventListener('message',()=>observedA++);b.webSocket.addEventListener('message',()=>observedB++);const r=await invoke(mf,A);assert.equal(r.status,200,await r.clone().text());const x=await r.json();assert.equal(x.result.label,'A');assert.equal(x.result.nonce,'1234567890abcdef1234567890abcdef');assert.equal(observedA,1);assert.equal(observedB,0);a.webSocket.close();b.webSocket.close();});
test('replacement connection survives stale close and no device means no dispatch',async t=>{const mf=await fixture(t);assert.equal((await invoke(mf,A)).status,503);const old=await open(mf,A,bindings.PROOF_DEVICE_A_TOKEN);old.webSocket.accept();const fresh=await open(mf,A,bindings.PROOF_DEVICE_A_TOKEN);responder(fresh.webSocket,'fresh');await new Promise(r=>setTimeout(r,10));const response=await invoke(mf,A);assert.equal(response.status,200);assert.equal((await response.json()).result.label,'fresh');fresh.webSocket.close();});
test('probe control route rejects missing credentials and malformed input',async t=>{const mf=await fixture(t);assert.equal((await mf.dispatchFetch('http://proof/api/invoke',{method:'POST',body:'{}'})).status,403);assert.equal((await mf.dispatchFetch('http://proof/api/invoke',{method:'POST',headers:{'x-localoud-proof-token':bindings.PROOF_CONTROL_TOKEN},body:'{}'})).status,400);});
